\section{Grundbedeutung und Wortart (Basic meaning and word class)}
\begin{grammarentry}{wie weit}{Interrogativadverbiale Wortverbindung (Lokaladverbial / Grad)}
\GermanText{\textbf{wie weit} fragt nach einer räumlichen Entfernung oder nach dem Ausmaß einer Entwicklung.}
\EnglishText{\textbf{wie weit} asks about a physical distance or the extent of a development.}

\end{grammarentry}
\section{Satzstellung und Verwendung (Word order and use)}
\begin{grammarentry}{Direkte und indirekte Frage}{Fragewortstellung}
\GermanText{In einer direkten W-Frage steht das finite Verb gewöhnlich an zweiter Stelle nach dem Fragewort. In einer indirekten Frage steht es am Ende. Die Bedeutung ergibt sich aus dem Kontext: räumliche Entfernung oder übertragener Fortschritt beziehungsweise Umfang.}
\EnglishText{In a direct information question, the finite verb normally follows the question word in second position. In an indirect question, it goes at the end. Context determines whether it refers to physical distance or figurative progress or extent.}
\end{grammarentry}
\begin{exbox}\begin{itemize}
\ExampleItem{Wie weit ist es bis zum Bahnhof?}{How far is it to the railway station?}
\ExampleItem{Wie weit ist die Untersuchung fortgeschritten?}{How far has the investigation progressed?}
\end{itemize}\end{exbox}
\section{Zusammenfassung (Summary)}
\begin{grammarentry}{Merksatz}{kurz und wichtig}
\GermanText{Merksatz: \textbf{wie weit} fragt nach Entfernung oder Ausmaß.}
\EnglishText{Summary: \textbf{wie weit} asks about distance or extent.}
\end{grammarentry}
