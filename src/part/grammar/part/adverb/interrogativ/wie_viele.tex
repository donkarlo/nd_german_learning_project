\section{Grundbedeutung und Wortart (Basic meaning and word class)}
\begin{grammarentry}{wie viel / wie viele}{Interrogativadverbiale Wortverbindung (Mengenfrage)}
\GermanText{\textbf{wie viel} fragt nach einer Menge; \textbf{wie viele} fragt nach der Anzahl zählbarer Dinge.}
\EnglishText{\textbf{wie viel} asks about an amount; \textbf{wie viele} asks about the number of countable things.}

\end{grammarentry}
\section{Satzstellung und Verwendung (Word order and use)}
\begin{grammarentry}{Direkte und indirekte Frage}{Fragewortstellung}
\GermanText{In einer direkten W-Frage steht das finite Verb gewöhnlich an zweiter Stelle nach dem Fragewort. In einer indirekten Frage steht es am Ende. \textbf{wie viele} steht gewöhnlich bei zählbaren Nomen im Plural: \textbf{Wie viele Bücher?}. \textbf{Wie viel} steht bei Mengen und nicht zählbaren Nomen: \textbf{Wie viel Wasser?}.}
\EnglishText{In a direct information question, the finite verb normally follows the question word in second position. In an indirect question, it goes at the end. \textbf{wie viele} normally accompanies countable plural nouns: \textbf{Wie viele Bücher?}. \textbf{Wie viel} is used for amounts and uncountable nouns: \textbf{Wie viel Wasser?}.}
\end{grammarentry}
\begin{exbox}\begin{itemize}
\ExampleItem{Wie viele Bücher hast du?}{How many books do you have?}
\ExampleItem{Wie viel Wasser trinkst du am Tag?}{How much water do you drink each day?}
\ExampleItem{Wie viel kostet das?}{How much does it cost?}
\end{itemize}\end{exbox}
\section{Zusammenfassung (Summary)}
\begin{grammarentry}{Merksatz}{kurz und wichtig}
\GermanText{Merksatz: \textbf{wie viel} fragt nach einer Menge; \textbf{wie viele} nach einer Anzahl.}
\EnglishText{Summary: \textbf{wie viel} asks about an amount; \textbf{wie viele} asks about a number.}
\end{grammarentry}
