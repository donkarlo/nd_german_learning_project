\section{Grundbedeutung und Wortart (Basic meaning and word class)}
\begin{grammarentry}{wie spät}{Interrogativadverbiale Wortverbindung (temporal)}
\GermanText{\textbf{wie spät} ist eine temporale Interrogativwortverbindung. Sie fragt nach der Uhrzeit.}
\EnglishText{\textbf{wie spät} is a temporal interrogative expression. It asks for the time of day.}

\end{grammarentry}
\section{Satzstellung und Verwendung (Word order and use)}
\begin{grammarentry}{Direkte und indirekte Frage}{Fragewortstellung}
\GermanText{In einer direkten W-Frage steht das finite Verb gewöhnlich an zweiter Stelle nach dem Fragewort. In einer indirekten Frage steht es am Ende. \textbf{Wie spät ist es?} ist eine feste und sehr häufige Frage nach der aktuellen Uhrzeit.}
\EnglishText{In a direct information question, the finite verb normally follows the question word in second position. In an indirect question, it goes at the end. \textbf{Wie spät ist es?} is a common fixed question asking for the current time.}
\end{grammarentry}
\begin{exbox}\begin{itemize}
\ExampleItem{Wie spät ist es? — Es ist halb acht.}{What time is it? — It is half past seven.}
\ExampleItem{Weißt du, wie spät der Laden schließt?}{Do you know what time the shop closes?}
\end{itemize}\end{exbox}

\EnglishText{Summary: \textbf{wie spät} asks for the time of day.}
