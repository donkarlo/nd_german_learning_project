\section{Grundbedeutung und Wortart (Basic meaning and word class)}
\begin{grammarentry}{wie oft}{Interrogativadverbiale Wortverbindung (temporal)}
\GermanText{\textbf{wie oft} ist eine temporale Interrogativwortverbindung. Sie fragt nach der Häufigkeit oder der Zahl der Wiederholungen.}
\EnglishText{\textbf{wie oft} is a temporal interrogative expression. It asks about frequency or the number of repetitions.}

\end{grammarentry}
\section{Satzstellung und Verwendung (Word order and use)}
\begin{grammarentry}{Direkte und indirekte Frage}{Fragewortstellung}
\GermanText{In einer direkten W-Frage steht das finite Verb gewöhnlich an zweiter Stelle nach dem Fragewort. In einer indirekten Frage steht es am Ende. Die Antwort nennt zum Beispiel \textbf{zweimal pro Woche}, \textbf{selten} oder \textbf{jeden Tag}.}
\EnglishText{In a direct information question, the finite verb normally follows the question word in second position. In an indirect question, it goes at the end. The answer may be, for example, \textbf{twice a week}, \textbf{rarely}, or \textbf{every day}.}
\end{grammarentry}
\begin{exbox}\begin{itemize}
\ExampleItem{Wie oft gehst du schwimmen?}{How often do you go swimming?}
\ExampleItem{Wie oft fährt der Bus am Abend?}{How often does the bus run in the evening?}
\end{itemize}\end{exbox}
\section{Zusammenfassung (Summary)}
\begin{grammarentry}{Merksatz}{kurz und wichtig}
\GermanText{Merksatz: \textbf{wie oft} fragt nach Häufigkeit.}
\EnglishText{Summary: \textbf{wie oft} asks about frequency.}
\end{grammarentry}
