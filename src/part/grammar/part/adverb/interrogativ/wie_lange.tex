\section{Grundbedeutung und Wortart (Basic meaning and word class)}
\begin{grammarentry}{wie lange}{Interrogativadverbiale Wortverbindung (temporal)}
\GermanText{\textbf{wie lange} ist eine temporale Interrogativwortverbindung. Sie fragt nach der Dauer: „über welchen Zeitraum?“.}
\EnglishText{\textbf{wie lange} is a temporal interrogative expression. It asks about duration: “for how long?”.}

\end{grammarentry}
\section{Satzstellung und Verwendung (Word order and use)}
\begin{grammarentry}{Direkte und indirekte Frage}{Fragewortstellung}
\GermanText{In einer direkten W-Frage steht das finite Verb gewöhnlich an zweiter Stelle nach dem Fragewort. In einer indirekten Frage steht es am Ende. Davon zu unterscheiden ist \textbf{seit wann}, das nach dem Beginn fragt.}
\EnglishText{In a direct information question, the finite verb normally follows the question word in second position. In an indirect question, it goes at the end. Distinguish it from \textbf{seit wann}, which asks when something began.}
\end{grammarentry}
\begin{exbox}\begin{itemize}
\ExampleItem{Wie lange dauert der Film?}{How long is the film?}
\ExampleItem{Wie lange wohnst du schon in Graz?}{How long have you been living in Graz?}
\end{itemize}\end{exbox}
\section{Zusammenfassung (Summary)}
\begin{grammarentry}{Merksatz}{kurz und wichtig}
\GermanText{Merksatz: \textbf{wie lange} fragt nach einer Dauer.}
\EnglishText{Summary: \textbf{wie lange} asks about a duration.}
\end{grammarentry}
