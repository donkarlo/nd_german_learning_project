\section{Grundbedeutung (Basic meaning)}
\begin{grammarentry}{hinter}{behind}
\GermanText{\textbf{hinter} bedeutet räumlich: auf der abgewandten Seite / hinter etwas.}
\EnglishText{\textbf{hinter} in spatial use means: on the far side / behind something.}
\end{grammarentry}
\section{Wo? + Dativ (Location: Where? + dative)}
\begin{grammarentry}{hinter + Dativ}{location}
\GermanText{Der Ausdruck bezeichnet die Position, an der sich jemand oder etwas befindet oder eine Handlung abspielt.}
\EnglishText{The expression describes the location where somebody or something is or where an action takes place.}
\end{grammarentry}
\begin{exbox}
\begin{itemize}
\ExampleItem{Der Hund wartet hinter dem Haus.}{The dog is waiting behind the house.}
\end{itemize}
\end{exbox}
\section{Wohin? + Akkusativ (Destination: Where to? + accusative)}
\begin{grammarentry}{hinter + Akkusativ}{destination}
\GermanText{Der Ausdruck bezeichnet den Zielort einer Bewegung oder Platzierung.}
\EnglishText{The expression describes the destination of a movement or placement.}
\end{grammarentry}
\begin{exbox}
\begin{itemize}
\ExampleItem{Der Hund läuft hinter das Haus.}{The dog runs to a position behind the house.}
\end{itemize}
\end{exbox}
\section{Übertragene und feste Verwendungen (Figurative and fixed uses)}
\begin{grammarentry}{Weitere Bedeutungen}{other uses}
\GermanText{\textbf{hinter jemandem stehen} kann bedeuten, jemanden zu unterstützen (Dativ). \textbf{hinter etwas kommen} kann bedeuten, etwas herauszufinden (Akkusativ).}
\EnglishText{\textbf{hinter jemandem stehen} can mean supporting somebody (dative). \textbf{hinter etwas kommen} can mean discovering something (accusative).}
\end{grammarentry}
\begin{exbox}
\begin{itemize}
\ExampleItem{Ich stehe hinter meiner Kollegin.}{I support my colleague.}
\ExampleItem{Sie ist hinter das Geheimnis gekommen.}{She discovered the secret.}
\end{itemize}
\end{exbox}
\section{Vergleich und häufiger Fehler (Comparison and common error)}
\begin{grammarentry}{Ort oder Ziel?}{location or destination}
\GermanText{\textbf{Der Hund wartet hinter dem Haus.} (Dativ, Wo?) im Gegensatz zu \textbf{Der Hund läuft hinter das Haus.} (Akkusativ, Wohin?). Nicht jedes Bewegungsverb verlangt den Akkusativ.}
\EnglishText{\textbf{Der Hund wartet hinter dem Haus.} (dative, location) contrasts with \textbf{Der Hund läuft hinter das Haus.} (accusative, destination). Movement verbs do not automatically require accusative.}
\end{grammarentry}
\section{Zusammenfassung (Summary)}
\begin{grammarentry}{Merksatz}{key rule}
\GermanText{Räumlich: \textbf{Wo? = Dativ, Wohin? = Akkusativ}. Die feste Rektion muss getrennt gelernt werden.}
\EnglishText{Spatial use: \textbf{Where? = dative, Where to? = accusative}. Learn fixed case government separately.}
\end{grammarentry}
